// Admin: block a student. hours = number of hours, or null = until an admin unblocks them
export async function adminBlockStudent(id, hours = null) {
  await wait();
  const a = admin();
  const u = userById(Number(id));
  if (!u) throw new ApiError('Student not found.', {}, 404);
  u.blocked_until = hours ? iso(Date.now() + hours * HOUR) : '2099-12-31T23:59:59.000Z';
  addLog('blocked', null, `${u.name} blocked by admin`, 0, null, { admin_id: a.id });
  return { message: hours ? `${u.name} is blocked.` : `${u.name} is blocked until you unblock them.`, blocked_until: u.blocked_until };
}
